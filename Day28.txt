package com.gl.service;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.web.client.RestTemplate;

import com.gl.beans.CourseCatalog;
import com.gl.beans.CourseRating;
import com.gl.beans.EnrolledCourse;
import com.gl.dao.CatalogDao;

@Service
public class CatalogServiceImpl implements CatalogService {
    
    @Autowired
    private CatalogDao dao;
    
    @Autowired
    private RestTemplate template;

    @Override
    public CourseCatalog getCatalogs(int userId) {
        
        CourseCatalog catalog = new CourseCatalog();
        catalog.setUserId(userId);
        
        List<CourseRating> aggregateRatings = new ArrayList<>();

        String enrollUrl = "http://localhost:8081/enroll/getcourses/" + userId;
        
        EnrolledCourse[] enrolledCourses = template.getForObject(enrollUrl, EnrolledCourse[].class);

        if (enrolledCourses != null) {
            for (EnrolledCourse en : enrolledCourses) {
                String ratingUrl = "http://localhost:8082/rating/getratings/" + en.getCourseId();
                
                CourseRating[] ratings = template.getForObject(ratingUrl, CourseRating[].class);
                
                if (ratings != null) {
                    aggregateRatings.addAll(Arrays.asList(ratings));
                }
            }
        }

        catalog.setRating(aggregateRatings);
        
        
        return catalog;
    }
}
